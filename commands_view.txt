#[tauri::command]
pub async fn parse_transaction(
    input: String, 
    policy: String, 
    partners: Vec<crate::core::models::Partner>
) -> Result<crate::core::models::AnalysisResponse, String> {
    println!("Frontend Request: {}", input);
    
    // Step 1: Journal AI Execution
    let parsed = ai_service::call_journal_ai(&input, &policy).await?;
    
    // Step 2: Compliance AI Review
    let compliance_review = ai_service::review_with_compliance(&parsed, &input).await.ok();
    
    let (vendor_status, suggested_vendor) = if let Some(ref reg_no) = parsed.vendor_reg_no {
        if let Some(matched) = crate::governance::vendor_governance::find_matching_partner(&partners, &Some(reg_no.clone())) {
            ("Matched".to_string(), Some(matched))
        } else {
            ("Pending_Registration".to_string(), Some(crate::governance::vendor_governance::create_pending_partner(&parsed)))
        }
    } else if let Some(ref vendor_name) = parsed.vendor {
        if vendor_name.is_empty() {
             ("No_Vendor".to_string(), None)
        } else {
            let matched_by_name = partners.iter()
                .find(|p| p.name == *vendor_name && p.status == "Approved")
                .cloned();
            
            if let Some(matched) = matched_by_name {
                ("Matched".to_string(), Some(matched))
            } else {
                ("Pending_Registration".to_string(), Some(crate::governance::vendor_governance::create_pending_partner(&parsed)))
            }
        }
    } else {
        ("No_Vendor".to_string(), None)
    };

    Ok(crate::core::models::AnalysisResponse {
        transaction: parsed,
        vendor_status,
        suggested_vendor,
        compliance_review,
    })
}

#[tauri::command]
pub async fn process_batch(csv_data: String) -> Result<Vec<ParsedTransaction>, String> {
    let mut transactions = crate::accounting::batch_processor::process_csv_batch(csv_data).await?;
    
    // AI Filtering Logic: Flag low confidence or risky items
    for tx in &mut transactions {
        let is_risky = tx.amount > 10_000_000.0 // High value asset/expense
            || tx.entry_type == "Grant" 
            || tx.vendor.as_ref().map_or(true, |v: &String| v.is_empty());

        if is_risky {
            tx.confidence = Some("Low".to_string());
            tx.needs_clarification = true;
            tx.clarification_prompt = Some("???泥섎━ 以?由ъ뒪?ш? 媛먯??섏뿀?듬땲?? ?곸꽭 ?뺤씤???꾩슂?⑸땲??".to_string());
        } else {
            tx.confidence = Some("High".to_string());
        }
    }
    
    Ok(transactions)
}

#[tauri::command]
pub fn approve_partner(mut partner: crate::core::models::Partner, partners: Vec<crate::core::models::Partner>) -> crate::core::models::Partner {
    partner.status = "Approved".to_string();
    partner.tags.retain(|t| t != "Hold Payment");
    
    // Generate a code (e.g., V10001)
    let count = partners.iter()
        .filter(|p| p.partner_type == "Vendor" && p.status == "Approved")
        .count();
    let code = format!("V{:05}", count + 1);
    partner.partner_code = Some(code);
    
    partner
}
